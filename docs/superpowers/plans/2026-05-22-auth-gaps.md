## Codebase Facts (read before touching anything)

- `AbstractRepository.paginate()` **already exists** in `lib/repositories/base.py:67–92`. Do not add a `list_paginated()` method — wire the existing one.
- `paginate()` returns `{"items": [serialized_dicts], "total": N, "page": P, "page_size": PS, "pages": max(1,...)}`. The `max(1,…)` is a bug — fix it in Task 3.
- `UserRepository._serialize()` calls `super()._serialize()` which reads **all** column attrs including `password_hash` and `salt`. Task 3 sanitizes the service output.
- `UserRepository.find_for_auth()` builds its dict manually — `is_active` must be added there explicitly (Task 1).
- `AbstractRepository.filter_by()` exists — use it in `request_reset()` to look up user by email.
- `lib.security.password.hash_password(plain)` returns `(hash, salt)` as strings.
- `UserService` dispatches methods by name via `SimpleService.execute()`. Methods that raise `ValueError` automatically become `ActionResult(success=False, error=str(exc))`.
- Tests use `_v1_app(*routers)` from `tests/conftest.py` + `TestClient` — no real server.
- `conftest.py` imports `lib.models` which imports everything in `lib/models/__init__.py` — adding a model there means every fixture picks it up for free.

```python
import secrets
import uuid
from datetime import datetime, timezone, timedelta

from sqlalchemy import and_

from lib.models.password_reset_token import PasswordResetToken
from lib.repositories.base import AbstractRepository


class PasswordResetTokenRepository(AbstractRepository[PasswordResetToken]):
    model = PasswordResetToken

    def create_for_user(self, user_id: uuid.UUID, ttl_seconds: int = 900) -> str:
        """Create a new reset token and return its raw string value."""
        token_str = secrets.token_hex(32)
        expires = datetime.now(timezone.utc) + timedelta(seconds=ttl_seconds)
        with self._factory.session() as s:
            s.add(PasswordResetToken(
                user_id=user_id,
                token=token_str,
                expires_at=expires,
            ))
        return token_str

    def find_valid(self, token: str) -> dict | None:
        """Return token row as dict if valid (not used, not expired). None otherwise."""
        now = datetime.now(timezone.utc)
        with self._factory.session() as s:
            row = (
                s.query(PasswordResetToken)
                .filter(
                    and_(
                        PasswordResetToken.token == token,
                        PasswordResetToken.used_at.is_(None),
                        PasswordResetToken.expires_at > now,
                    )
                )
                .first()
            )
            if row is None:
                return None
            return {
                "id": str(row.id),
                "user_id": str(row.user_id),
                "token": row.token,
            }

    def mark_used(self, token_id: uuid.UUID) -> None:
```

```python
    def register(self, data: dict) -> dict:
        """Public registration. Raises ValueError on duplicate username or email."""
        try:
            return self.create(data)
        except IntegrityError as exc:
            msg = str(exc).lower()
            if "username" in msg:
                raise ValueError("Username already registered") from exc
            if "email" in msg:
                raise ValueError("Email already registered") from exc
            raise

    def request_reset(self, data: dict) -> dict:
        """Request a password reset token for the given email address.

        Returns {"token": ..., "expires_in": 900} on success.
        Returns a generic message if the email is not registered (prevents user enumeration).
        """
        from lib.repositories.password_reset_token_repository import (
            PasswordResetTokenRepository,
        )
        repo = UserRepository(self._factory)
        token_repo = PasswordResetTokenRepository(self._factory)

        users = repo.filter_by(email=data["email"])
        if not users:
            return {"message": "If that email is registered, a reset token has been issued"}

        user_id = uuid.UUID(users[0]["id"])
        token_str = token_repo.create_for_user(user_id)
        return {"token": token_str, "expires_in": 900}

    def reset_password(self, data: dict) -> dict:
        """Reset a user's password using a valid reset token.

        Raises ValueError if the token is invalid or expired.
        """
        from lib.repositories.password_reset_token_repository import (
            PasswordResetTokenRepository,
        )
        token_repo = PasswordResetTokenRepository(self._factory)
        token_row = token_repo.find_valid(data["token"])
        if token_row is None:
            raise ValueError("Invalid or expired reset token")

        password_hash, salt = hash_password(data["new_password"])
        user_repo = UserRepository(self._factory)
        user_repo.update(uuid.UUID(token_row["user_id"]), {
            "password_hash": password_hash,
            "salt": salt,
        })
        token_repo.mark_used(uuid.UUID(token_row["id"]))
        return {"success": True}
```
